// Runs inside the external wow-addons-sandbox checkout, never in the shipped app.
import { Runtime } from './src/runtime';
import { Renderer } from './src/render';
import { loadFromFiles } from './src/addons';
import './src/style.css';

const errors: string[] = [];
const runtime = new Runtime({
  log(kind, text) {
    if (kind === 'error') errors.push(text);
    console[kind === 'error' ? 'error' : 'log'](text);
  },
  requestReload() {},
});
const data = await (await fetch('/wisp-addon.json')).json();
const rawFiles = Object.entries(data.sources).map(([path, source]) => ({
  path, lastModified: 0, size: String(source).length,
  async read() { return new TextEncoder().encode(String(source)).buffer; },
}));
rawFiles.push({ path: 'Textures/Gear.tga', lastModified: 0, size: 0,
  async read() { return (await fetch('/gear.tga')).arrayBuffer(); },
});
const [addon] = await loadFromFiles('WispAI', rawFiles);
const stage = document.getElementById('stage')!;
const renderer = new Renderer(stage, {
  mouse: (id, event, button, extra) => runtime.mouse(id, event, button, extra),
  edit: (id, event, text) => runtime.edit(id, event, text),
  cursor: (x, y) => runtime.mouseMove(x, y),
}, (path) => path.toLowerCase().includes('gear') ? addon.images.get('textures/gear') : undefined);
await runtime.boot([addon], { w: 1280, h: 720, sv: {}, client: 'forever' });
let previous = performance.now();
function render(now: number) {
  runtime.tick(Math.min((now - previous) / 1000, 0.05));
  previous = now;
  const snapshot = runtime.snapshot();
  if (snapshot) renderer.update(snapshot);
  requestAnimationFrame(render);
}
requestAnimationFrame(render);
Object.assign(window, {
  wispDemo: {
    ready: true,
    errors,
    stage(index: number) { runtime.slash(`/run WispDemoStage(${index})`); },
    run(code: string) { runtime.slash(`/run ${code}`); },
    snapshot() { return runtime.snapshot(); },
    caption(title: string) { document.getElementById('scene-title')!.textContent = title; },
  },
});
if (new URLSearchParams(location.search).has('controls')) {
  document.body.classList.add('interactive');
  for (const button of document.querySelectorAll<HTMLButtonElement>('[data-scene]')) {
    button.disabled = false;
    button.addEventListener('click', () => {
      runtime.slash(`/run WispDemoStage(${Number(button.dataset.scene)})`);
    });
  }
}
